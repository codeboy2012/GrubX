Required Parameters
{id}Movie identifier from IMDB or TMDB
Optional Parameters
title Controls whether the media title is displayed
poster Determines if the poster image is shown
autoPlay Controls whether the media starts playing automatically

theme Changes the player's color (hex code format)

Events & Progress Tracking
The player can send watch progress events to the parent window. You can save this progress to localStorage or your own backend.

Available Events
playTriggered when video starts playing
pauseTriggered when video is paused
seekedTriggered when user seeks to a different timestamp
endedTriggered when video playback ends
timeupdateTriggered periodically during playback
playerstatusTriggered when getStatus is called
Event Data Structure
{
    type: "PLAYER_EVENT",
    data: {
        event: "play" | "pause" | "seeked" | "ended" | "timeupdate" | "playerstatus",
        currentTime: number,
        duration: number,
        tmdbId: number,
        mediaType: "movie" | "tv",
        season?: number,
        episode?: number,
        playing: bool,
        muted: bool,
        volume: number
    }
}

Event Listener Implementation
Add this script where your iframe is located. For React/Next.js, place it in a useEffect hook.

const vidfastOrigins = [
    'https://vidfast.pro',
    'https://vidfast.in',
    'https://vidfast.io',
    'https://vidfast.me',
    'https://vidfast.net',
    'https://vidfast.pm',
    'https://vidfast.vc',
    'https://vidfast.bz'
];

window.addEventListener('message', ({ origin, data }) => {
    if (!vidfastOrigins.includes(origin) || !data) {
        return;
    }

    if (data.type === 'PLAYER_EVENT') {
        const { event, currentTime, duration } = data.data;

        console.log(`Player ${event} at ${currentTime}s of ${duration}s`);

        // Add custom event handling logic here
    }
});

PostMessage API Control
Control VidFast players programmatically using the PostMessage API. Perfect for watch party features and custom player integrations.

Available Commands
play
Resume video playback
iframe.contentWindow.postMessage({
    command: 'play'
}, '*');
pause
Pause video playback
iframe.contentWindow.postMessage({
    command: 'pause'
}, '*');
seek
Jump to specific time in video (seconds)
iframe.contentWindow.postMessage({
    command: 'seek',
    time: 120  // Jump to 2 minutes
}, '*');
volume
Set player volume (0.0 to 1.0)
iframe.contentWindow.postMessage({
    command: 'volume',
    level: 0.5  // Set to 50% volume
}, '*');
mute
Toggle mute state
iframe.contentWindow.postMessage({
    command: 'mute',
    muted: true  // true to mute, false to unmute
}, '*');
getStatus
Get current player status
iframe.contentWindow.postMessage({
    command: 'getStatus'
}, '*');

// Listen for response
window.addEventListener('message', ({ data }) => {
    if (data.type === 'PLAYER_EVENT' && data.data.event === 'playerstatus') {
        console.log('Current time:', data.data.currentTime);
        console.log('Duration:', data.data.duration);
        console.log('Is playing:', data.data.playing);
        console.log('Is muted:', data.data.muted);
        console.log('Volume:', data.data.volume);
    }
});
Watch Party Integration Example
Perfect for synchronizing video playback across multiple users in a watch party scenario.

// Watch Party Controller Example
class WatchPartyController {
    vidfastOrigins = [
        'https://vidfast.pro',
        'https://vidfast.in',
        'https://vidfast.io',
        'https://vidfast.me',
        'https://vidfast.net',
        'https://vidfast.pm',
        'https://vidfast.xyz',
        'https://vidfast.vc',
        'https://vidfast.bz'
    ]

    constructor(iframeElement) {
        this.iframe = iframeElement;
        this.setupEventListeners();
    }

    // Sync play command to all participants
    syncPlay(time) {
        this.iframe.contentWindow.postMessage({
            command: 'play',
            time: time
        }, '*');

        // Broadcast to other participants
        this.broadcastToParty({
            action: 'play',
            time: time
        });
    }

    // Sync pause command to all participants
    syncPause(time) {
        this.iframe.contentWindow.postMessage({
            command: 'pause',
            time: time
        }, '*');

        this.broadcastToParty({
            action: 'pause',
            time: time
        });
    }

    // Sync seek to specific time for all participants
    syncSeek(time) {
        this.iframe.contentWindow.postMessage({
            command: 'seek',
            time: time
        }, '*');

        this.broadcastToParty({
            action: 'seek',
            time: time
        });
    }

    // Handle incoming party commands
    handlePartyCommand(command) {
        switch (command.action) {
            case 'play':
                this.iframe.contentWindow.postMessage({
                    command: 'play'
                }, '*');
                break;
            case 'pause':
                this.iframe.contentWindow.postMessage({
                    command: 'pause'
                }, '*');
                break;
            case 'seek':
                this.iframe.contentWindow.postMessage({
                    command: 'seek',
                    time: command.time
                }, '*');
                break;
        }
    }

    broadcastToParty(command) {
        // Your party synchronization logic here
        // (WebSocket, Socket.IO, etc.)
    }

    onPlayerStatusUpdate(status) {
        // Your status update logic here
    }

    setupEventListeners() {
        // Listen for player events
        window.addEventListener('message', (event) => {
            if (!this.vidfastOrigins.includes(event.origin) || !event.data) {
                return;
            }

            if (event.data.type === 'PLAYER_EVENT') {
                const {
                    event: playerEvent,
                    currentTime
                } = event.data.data;

                switch (playerEvent) {
                    case 'play':
                        this.syncPlay(currentTime);
                        break;
                    case 'pause':
                        this.syncPause(currentTime);
                        break;
                    case 'seeked':
                        this.syncSeek(currentTime);
                        break;
                }
            }

            if (event.data.type === 'PLAYER_EVENT' && event.data.data.event === 'playerstatus') {
                this.onPlayerStatusUpdate(event.data.data);
            }
        });
    }
}

// Usage
const iframe = document.querySelector('#vidfast-player');
const watchParty = new WatchPartyController(iframe);
⚠️ Important Notes
• PostMessage commands work across all VidFast domains
• Commands are processed asynchronously - status responses may have a small delay
• For watch parties, implement proper synchronization logic to handle network latency
• Always handle message events to receive player status updates
• Seek commands accept time in seconds (integer values)

Customization Parameters
You can customize the embedded media player by appending parameters to the URL. Each parameter should start with a ? and multiple parameters should be separated by &.
Use Hex Color Codesand remove the '#' before applying.
primaryColor
Sets the primary color of the player, including sliders and autoplay controls.

primaryColor=B20710
secondaryColor
Defines the color of the progress bar behind the sliders.

secondaryColor=170000
icons
Changes the design of the icons within the player. can be either "vid" or "default".

icons=vid
Example of "vid" icons:
iconColor
Changes the color of the icons within the player.

iconColor=B20710
title
Controls whether the media title is displayed.

title=false
poster
Determines if the poster image is shown.

poster=true
autoplay
Controls whether the media starts playing automatically.

autoplay=false
Next Episode button
Shows next episode button when 90% of the Tv-show is watched. OFF by default.

nextbutton=true
Player typeNew
Changes the player to JWPlayer or default player.

player=jw
startAtNew
Starts the video at the specified time in seconds. This parameter cannot replace saved progress but can be used for cross-device watch progress. remove cookies and cache after each test for the same content.

startAt=60
sub_fileNew
Adds external subtitles to the video. Must be a direct link to a VTT subtitle file.

sub_file=https://example.com/subtitles.vtt
sub_labelNew
Sets the label for the external subtitle track. If not provided, defaults to 'External Subtitle'.

sub_label=English
fallback_urlNew
Redirect URL when stream fails to load. If the stream is unavailable, the player will redirect to this URL.

fallback_url=https://example.com/fallback

theme
string
default: default
Changes the player accent color. Accepts a hex color code without the # prefix (e.g. theme=16A085)

title
boolean
default: true
Shows or hides the media title overlay in the player

poster
boolean
default: true
Shows or hides the poster/thumbnail image before playback begins

autoPlay
boolean
default: false
Starts playback automatically when the player loads. May be blocked by browser policies

startAt
number
default: 0
Starts the video at a specific time. Value is in seconds (e.g. startAt=120 starts at 2 minutes)

sub
string
default: none
Sets the default subtitle track using language codes (e.g. en, es, fr, de, pt)

server
string
default: auto
Sets the default streaming server. Use the exact server name as it appears in the server selector

hideServer
boolean
default: false
Hides the server selector button from the player controls

fullscreenButton
boolean
default: true
Shows or hides the fullscreen toggle button in the player

chromecast
boolean
default: true
Shows or hides the Chromecast/AirPlay button in the player controls

nextButton
boolean
default: false
TV only — Shows a 'Next Episode' button when 90% of the current episode has been watched

autoNext
boolean
default: false
TV only — Automatically advances to the next episode when the current one finishes. Requires nextButton=true

Watch Party Integration Example
Perfect for synchronizing video playback across multiple users in a watch party scenario.

Sync Example
Copy
// Watch Party Controller Example
class WatchPartyController {
    vidcoreOrigins = [
        'https://vidcore.io'
    ]

    constructor(iframeElement) {
        this.iframe = iframeElement;
        this.setupEventListeners();
    }

    // Sync play command to all participants
    syncPlay(time) {
        this.iframe.contentWindow.postMessage({
            command: 'play',
            time: time
        }, '*');

        // Broadcast to other participants
        this.broadcastToParty({
            action: 'play',
            time: time
        });
    }

    // Sync pause command to all participants
    syncPause(time) {
        this.iframe.contentWindow.postMessage({
            command: 'pause',
            time: time
        }, '*');

        this.broadcastToParty({
            action: 'pause',
            time: time
        });
    }

    // Sync seek to specific time for all participants
    syncSeek(time) {
        this.iframe.contentWindow.postMessage({
            command: 'seek',
            time: time
        }, '*');

        this.broadcastToParty({
            action: 'seek',
            time: time
        });
    }

    // Handle incoming party commands
    handlePartyCommand(command) {
        switch (command.action) {
            case 'play':
                this.iframe.contentWindow.postMessage({
                    command: 'play'
                }, '*');
                break;
            case 'pause':
                this.iframe.contentWindow.postMessage({
                    command: 'pause'
                }, '*');
                break;
            case 'seek':
                this.iframe.contentWindow.postMessage({
                    command: 'seek',
                    time: command.time
                }, '*');
                break;
        }
    }

    broadcastToParty(command) {
        // Your party synchronization logic here
        // (WebSocket, Socket.IO, etc.)
    }

    onPlayerStatusUpdate(status) {
        // Your status update logic here
    }

    setupEventListeners() {
        // Listen for player events
        window.addEventListener('message', (event) => {
            if (!this.vidcoreOrigins.includes(event.origin) || !event.data) {
                return;
            }

            if (event.data.type === 'PLAYER_EVENT') {
                const {
                    event: playerEvent,
                    currentTime
                } = event.data.data;

                switch (playerEvent) {
                    case 'play':
                        this.syncPlay(currentTime);
                        break;
                    case 'pause':
                        this.syncPause(currentTime);
                        break;
                    case 'seeked':
                        this.syncSeek(currentTime);
                        break;
                }
            }

            if (event.data.type === 'PLAYER_EVENT' && event.data.data.event === 'playerstatus') {
                this.onPlayerStatusUpdate(event.data.data);
            }
        });
    }
}

// Usage
const iframe = document.querySelector('#vidcore-player');
const watchParty = new WatchPartyController(iframe);

06
JavaScript SDK
https://ani.pm/embed/sdk/v1.js is dependency-free (about 5 KB compressed), defines window.AniPM and also works with CommonJS. Every method returns a Promise that resolves with the player's state, or rejects with an AniPMError whose code says why. Calls made before the embed has loaded are queued.

Playback starts on its own. The player starts as soon as it loads; a browser that blocks sound starts it muted, and the viewer unmutes with one tap. With autoplay=0 the Play screen waits for the viewer: until then loadEpisode(), next(), previous(), seek(), setLanguage() and setHardsub() choose what it will play, start() or play() starts it, and state.awaitingPress is true.

play · pause · playing · waiting	Playback starts, pauses, resumes after buffering, or buffers.
seeking · seeked	A jump starts and lands.
timeupdate	Once a second while playing: { currentTime, duration, paused }.
durationchange · ratechange · volumechange	Media changes.
ended	The episode finished (the next one follows when autoNext is on).
episodechange · languagechange	Another episode, or sub/dub or hard/soft changed.
episodes	The title's episode list arrived.
qualitychange · subtitlechange	Rendition or subtitle track changed, with the lists.
fullscreenchange	The player entered or left fullscreen.
autonextchange · autoskipchange	The viewer flipped a switch inside the player.
theaterchange · ambientchange	{ theater } or { ambient }: theater mode or the ambient glow turned on or off.
settings	The viewer’s saved settings changed ({ settings }); the SDK keeps this copy for you.
autoplayblocked	The browser refused to start playback without a click.
error	{ code, message }: see Errors.
08
postMessage protocol
The SDK is a thin client over messages you can send yourself. Messages are objects { ns: 'anipm.player', v: 1, … }, exchanged between the page that frames the embed and the embed itself; check event.origin (https://ani.pm) and event.source on your side.

Direction	Shape
page → embed	{ ns, v: 1, id?: number, cmd: string, args?: object }
embed → page (reply)	{ ns, v: 1, id, ok: true, result }
embed → page (failure)	{ ns, v: 1, id, ok: false, error: { code, message } }
embed → page (event)	{ ns, v: 1, event: string, data }
Commands: hello (send first; the reply carries the state), getState, getEpisodes, start, play, pause, togglePlay, seek { time }, setVolume { volume, muted? }, setMuted { muted }, setPlaybackRate { rate }, setSubtitles { enabled, label?, language? }, setQuality { height }, setLanguage { lang }, setHardsub { on }, loadEpisode { episode, lang?, start? }, next, previous, setAutoNext { on }, setAutoSkip { on }, setTheater { on }, toggleTheater, setAmbient { on }, setAccent { color }. hello may carry { settings }: a copy of the settings event’s data, which the embed takes back when its own storage has lost it. At most 40 commands a second.

Talking to the embed without the SDK
Copy
const frame = document.querySelector('iframe')
const ORIGIN = 'https://ani.pm'
let id = 0
const pending = new Map()

function send(cmd, args = {}) {
  return new Promise((resolve, reject) => {
    const n = ++id
    pending.set(n, { resolve, reject })
    frame.contentWindow.postMessage({ ns: 'anipm.player', v: 1, id: n, cmd, args }, ORIGIN)
  })
}

window.addEventListener('message', (event) => {
  if (event.source !== frame.contentWindow || event.origin !== ORIGIN) return
  const m = event.data
  if (!m || m.ns !== 'anipm.player' || m.v !== 1) return
  if (m.id && pending.has(m.id)) {                  // a reply
    const p = pending.get(m.id); pending.delete(m.id)
    m.ok ? p.resolve(m.result) : p.reject(Object.assign(new Error(m.error.message), { code: m.error.code }))
  } else if (m.event) {                             // an event
    console.log(m.event, m.data)
  }
})

frame.addEventListener('load', () => send('hello').then((r) => console.log('connected', r.state)))

Apps and keys
Everything above works without one. An app (free, at Developers › Your apps, signed in) adds:

A full analytics page for each app (Analytics in the dashboard, and /analytics/report,/analytics/plays and /analytics/realtime for your server): every play and what happened in it; live playback as it is (each open player, its position, playing, paused or buffering, and its settings); plays, viewers, watch time, completion and the furthest point reached against the period before; retention; when people watch; titles and episodes; countries, languages and time zones; sites and pages; devices, browsers and systems with versions; startup, rebuffering and stalls; audio, subtitles, resolution, speed, fullscreen and theater; how plays start and end; problems; your REST calls with status and latency. Any range up to 400 days in your own time zone, every row a filter, exportable as CSV or JSON. Your apps are listed side by side there too.
Your defaults: audio language, next episode, skip intros, accent color, 18+ on or off, for every embed that carries your key (/embed/v1/{anilistId}/{episode}?key=pk_live_…).
A site lock, if you want one: list your sites and your key's embeds stay blank anywhere else. An app that lists no sites works everywhere.
A secret key for your server (X-API-Key: sk_live_…): your usage and analytics over the REST API. It is shown once; never put it in front-end code (browsers that send it are refused).
What analytics collect. Plays, embeds shown and API calls are counted where ani.pm serves them; everything that happens in a play (watch time, the furthest point, startup, stalls, seeks, pauses, resolution, subtitles, speed, fullscreen, sound, and where it is while it plays) comes from the player. No address, user agent or full page address is stored: a viewer is an anonymous tag from a hash whose daily key is discarded after two days (so it cannot be tied to anyone, even by ani.pm), countries come from each viewer’s connection, sites are the host names of the pages the player is on, pages are the paths the SDK reports (never a query string; turn it off with reportPage: false), and devices, browsers and systems are classes and major versions. Numbers arrive within about 15 seconds; live players report every 15 seconds. Plays are kept for 400 days, and an app's analytics are deleted with the app.

11
Security
Every stream is private to its viewer. Playback runs on short-lived sessions signed for one episode, inside ani.pm’s frames; your page never sees a media URL, and there is nothing to hotlink.
Framing is signed. The player frame accepts the page around it only as ani.pm signs it for that one session: any site for an open embed, only your sites for a locked app.
Only a frame plays. An embed link opened in a tab goes to the title on ani.pm.
No viewer data crosses between your page and ani.pm. The embed reads no ani.pm account and sends your page only the player events listed here; the player remembers its own settings (volume, subtitles) per site.
Your page’s CSP: allow frame-src https://ani.pm, and script-src https://ani.pm if you use the SDK.
Do not sandbox the iframe. The player needs scripts, its own origin, popups and presentation; a sandbox without them stops it, and the embed says so.
12
Fair use
The player opens at most 1 popover every 5 minutes. Show it as ani.pm serves it:

Do not cover, shrink below 320×180, hide or intercept clicks on the player.
Do not frame the player inside ad-stacking or click-generating pages.
Sites and apps that break these rules lose the player without notice.

13
No limits
There are none. No key, no signup, no quotas and no rate limits: plays per viewer, per site and per day, REST requests with or without a key, and SDK commands are all unlimited, for open embeds and for apps alike.

What stays is abuse protection, not a quota. ani.pm can switch off an app or a site that misuses the player (see Developer terms), and its network edge turns away floods far beyond anything a page or a server sends.
Cache what you can. REST answers carry Cache-Control; honouring it makes your pages faster, it is not required.

  qualitychange: { quality: PlayerState['quality']; qualities: Quality[] }
  subtitlechange: { subtitles: PlayerState['subtitles']; tracks: SubtitleTrack[] }
  fullscreenchange: { fullscreen: boolean }
  autonextchange: { autoNext: boolean }
  autoskipchange: { autoSkip: boolean }
  autoplayblocked: PlayerState
  theaterchange: { theater: boolean }
  ambientchange: { ambient: boolean }
  /** The viewer's saved settings changed (the SDK keeps this copy for you unless rememberSettings is false). */
  settings: { settings: unknown }
}
export type PlayerEventName = keyof PlayerEventMap

export declare class AniPMError extends Error {
  readonly code: string
}

  constructor(target: string | Element, options: PlayerOptions)
  static readonly version: string
  readonly iframe: HTMLIFrameElement

  readonly state: PlayerState

  once<K extends PlayerEventName>(event: K, listener: (data: PlayerEventMap[K], player: Player) => void): () => void
  /** Resolves when the embed answers (not when video plays; see the 'ready' event). */
  ready(): Promise<PlayerState>
  /** Starts playback as if Play was pressed (an autoplay: false player waits for this or for the viewer). */
  start(): Promise<PlayerState>
  /** Plays; starts the player first if it has not started yet. */
  play(): Promise<PlayerState>
  pause(): Promise<PlayerState>
  togglePlay(): Promise<PlayerState>
  seek(seconds: number): Promise<PlayerState>
  seekBy(deltaSeconds: number): Promise<PlayerState>
  setVolume(volume: number): Promise<PlayerState>
  setMuted(muted: boolean): Promise<PlayerState>
  mute(): Promise<PlayerState>
  unmute(): Promise<PlayerState>
  setPlaybackRate(rate: number): Promise<PlayerState>
  setSubtitles(choice: false | 'off' | string | { label?: string; language?: string }): Promise<PlayerState>
  /** A picture height from state.qualities. There is no automatic quality. */
  setQuality(height: number): Promise<PlayerState>
  setLanguage(lang: Language): Promise<PlayerState>
  setHardsub(on: boolean): Promise<PlayerState>
  loadEpisode(episode: number, options?: { lang?: Language; start?: number }): Promise<PlayerState>
  next(): Promise<PlayerState>
  previous(): Promise<PlayerState>
  setAutoNext(on: boolean): Promise<PlayerState>
  setAutoSkip(on: boolean): Promise<PlayerState>
  setTheater(on: boolean): Promise<PlayerState>
  toggleTheater(): Promise<PlayerState>
  /** The ambient glow (shows in theater and fullscreen). Remembered by the player on this device. */

  setAccent(color: string): Promise<PlayerState>
  getState(): Promise<PlayerState>
  getEpisodes(): Promise<Episode[]>
  loadTitle(anime: number | string, episode?: number, options?: Partial<PlayerOptions>): Promise<PlayerState>
  /** Call from your own click handler. */
  requestFullscreen(): Promise<void>
  exitFullscreen(): Promise<void>
  destroy(): void
}
